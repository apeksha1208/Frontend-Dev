// 1. Click manager → highlight reports $(".manager").click(function(){ $(this).children(".employee").css("background","lightblue"); }); // 2. Hover employee → show contact $(".employee").hover(function(){ $(this).next(".contact").toggle(); }); // 3. Click department → highlight members $(".department").click(function(){ $(this).children(".member").css("background","lightgreen"); }); // 4. Random employee highlight siblings let rand = Math.floor(Math.random() * $(".employee").length); let emp = $(".employee").eq(rand); emp.siblings().css("border","2px solid red"); // 5. Collapse/expand team $(".toggleTeam").click(function(){ $(this).parent().find(".employee").toggle(); });
