// Toggle answer visibility $(".question").click(function(){ $(this).next(".answer").toggle(); }); // Hover effect $(".question").hover( function(){ $(this).css("color","blue"); }, function(){ $(this).css("color","black"); } ); // Double click collapse all $(".question").dblclick(function(){ $(".answer").hide(); }); // Focus on input inside answer $("input").focus(function(){ $(this).closest(".faq-item").css("background","lightyellow"); }); // Blur → reset $("input").blur(function(){ $(this).closest(".faq-item").css("background","white"); });
